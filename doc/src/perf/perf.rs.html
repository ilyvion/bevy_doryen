7#[derive(Resource)]
8struct PerfTest {
9    seed: u64,
10}
11
12impl PerfTest {
13    pub fn new() -> Self {
14        Self { seed: 0xdead_beef }
15    }
16    fn rnd(&mut self) -> u64 {
17        self.seed = 214_013u64.wrapping_mul(self.seed).wrapping_add(2_531_011);
18        self.seed
19    }
20}
21
22pub fn main() {
23    App::new()
24        .insert_resource(DoryenPluginSettings {
25            app_options: AppOptions {
26                window_title: String::from("bevy_doryen performance test"),
27                vsync: false,
28                ..Default::default()
29            },
30            resize_mode: ResizeMode::Automatic,
31            ..Default::default()
32        })
33        .add_plugins(DoryenPlugin)
34        .insert_resource(PerfTest::new())
35        .add_systems(Render, render)
36        .run();
37}

39fn render(
40    mut root_console: ResMut<RootConsole>,
41    mut perf_test: ResMut<PerfTest>,
42    fps_info: Res<FpsInfo>,
43) {
44    let fps = fps_info.fps;
45    let con_width = root_console.get_width();
46    let con_height = root_console.get_height();
47    for y in 0..con_height as i32 {
48        for x in 0..con_width as i32 {
49            let val = perf_test.rnd();
50            root_console.back(
51                x,
52                y,
53                (
54                    (val & 0xFF) as u8,
55                    ((val >> 8) & 0x5F) as u8,
56                    ((val >> 16) & 0x3F) as u8,
57                    255,
58                ),
59            );
60            root_console.fore(
61                x,
62                y,
63                (
64                    ((val >> 16) & 0xFF) as u8,
65                    ((val >> 24) & 0xFF) as u8,
66                    ((val >> 32) & 0xFF) as u8,
67                    255,
68                ),
69            );
70            root_console.ascii(x, y, ((val >> 40) & 0xFF) as u16);
71        }
72    }
73    root_console.rectangle(
74        (con_width / 2 - 10) as i32,
75        (con_height / 2 - 2) as i32,
76        20,
77        5,
78        Some((255, 255, 255, 255)),
79        Some((0, 0, 0, 255)),
80        Some(' ' as u16),
81    );
82    root_console.print(
83        (con_width / 2) as i32,
84        (con_height / 2) as i32,
85        &format!("{} fps", fps),
86        TextAlign::Center,
87        Some((255, 255, 255, 255)),
88        None,
89    );
90}
