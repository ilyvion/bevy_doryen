1use std::fmt;
2
3use doryen_rs::AppOptions;
4
5// Because `AppOptions` doesn't support `Debug`, but I would like to, I'm just
6// going to do it myself.
7pub(crate) struct DebugAppOptions<'a>(pub(crate) &'a AppOptions);
8impl fmt::Debug for DebugAppOptions<'_> {
9    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
10        let AppOptions {
11            console_width,
12            console_height,
13            screen_width,
14            screen_height,
15            window_title,
16            font_path,
17            vsync,
18            fullscreen,
19            show_cursor,
20            resizable,
21            intercept_close_request,
22            max_fps,
23            ..
24        } = self.0;
25        f.debug_struct("AppOptions")
26            .field("console_width", console_width)
27            .field("console_height", console_height)
28            .field("screen_width", screen_width)
29            .field("screen_height", screen_height)
30            .field("window_title", window_title)
31            .field("font_path", font_path)
32            .field("vsync", vsync)
33            .field("fullscreen", fullscreen)
34            .field("show_cursor", show_cursor)
35            .field("resizable", resizable)
36            .field("intercept_close_request", intercept_close_request)
37            .field("max_fps", max_fps)
38            .finish()
39    }
40}
