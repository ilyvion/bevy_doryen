1//! bevy_doryen is a Bevy plugin that lets you use [Bevy] with the [Doryen]
2//! roguelike library.
3//!
4//! Usage:
5//! ```no_run
6//! # use bevy_app::{App, Update, Startup};
7//! # use bevy_doryen::prelude::*;
8//! # use bevy_doryen::{ResizeMode, MouseButton};
9//! # use bevy_doryen::doryen::AppOptions;
10//! # use bevy_ecs::system::IntoSystem;
11//! App::new()
12//!     // Insert a `DoryenPluginSettings` resource to configure the plugin.
13//!     .insert_resource(DoryenPluginSettings {
14//!         // `app_options` lets you configure Doryen just as if you were
15//!         // using Doryen without Bevy. The default is `AppOptions::default()`.
16//!         app_options: AppOptions {
17//!             show_cursor: true,
18//!             resizable: true,
19//!             ..AppOptions::default()
20//!         },
21//!         // Lets you configure which mouse buttons to listen for. The default
22//!         // is left, middle and right click.
23//!         mouse_button_listeners: vec![
24//!             MouseButton::Left,
25//!             MouseButton::Middle,
26//!             MouseButton::Right,
27//!         ],
28//!         // Lets you configure how the application should behave when resized.
29//!         // The default is `ResizeMode::Nothing`. See `ResizeMode`'s
30//!         // documentation for more information.
31//!         resize_mode: ResizeMode::Nothing
32//!     })
33//!     // Add the `DoryenPlugin` to Bevy.
34//!     .add_plugins(DoryenPlugin)
35//!     // Add your Bevy systems like usual. Excluding startup systems, which
36//!     // only run once, these systems are run during Doryen's update phase;
37//!     // i.e. 60 times per second.
38//!     .add_systems(Startup, init)
39//!     .add_systems(Update, input)
40//!     // The `Render` schedules lets you add systems that should
41//!     // be run during Doryen's render phase.
42//!     .add_systems(Render, render)
43//!     .run();
44//!
45//! # fn init() { }
46//! # fn input() { }
47//! # fn render() { }
48//! ```
49//!
50//! [Bevy]: https://bevyengine.org/
51//! [Doryen]: https://github.com/jice-nospam/doryen-rs

148/// DoryenPlugin settings.
149#[derive(Resource)]
150pub struct DoryenPluginSettings {
151    /// The [`AppOptions`] passed to the [`DoryenApp`].
152    pub app_options: AppOptions,
153    /// Which mouse buttons to request input data for from Doryen during the
154    /// input handling.
155    /// Defaults to left, middle and right mouse buttons.
156    pub mouse_button_listeners: Vec<MouseButton>,
157    /// What to do when the Doryen window is resized.
158    pub resize_mode: ResizeMode,
159}

188impl Plugin for DoryenPlugin {
189    fn build(&self, app: &mut BevyApp) {
190        app.init_resource::<RootConsole>()
191            .init_resource::<Input>()
192            .init_resource::<FpsInfo>()
193            .add_event::<SetFontPath>()
194            .add_event::<Resized>()
195            .add_event::<Capture>()
196            .add_plugins(RenderSchedulePlugin)
197            .set_runner(doryen_runner);
198    }

201struct DoryenPluginEngine {
202    bevy_app: BevyApp,
203    app_exit_event_reader: ManualEventReader<AppExit>,
204    set_font_path_event_reader: ManualEventReader<SetFontPath>,
205    capture_event_reader: ManualEventReader<Capture>,
206    swap_console: Option<Console>,
207    mouse_button_listeners: Vec<MouseButton>,
208    previous_screen_size: (u32, u32),
209    previous_console_size: (u32, u32),
210    resize_mode: ResizeMode,
211}
212
213impl DoryenPluginEngine {
214    #[inline]
215    fn take_root_console_ownership(&mut self, api: &mut dyn DoryenApi) {
216        use std::mem::swap;
217
218        // Take ownership of the Doryen root console
219        swap(api.con(), self.swap_console.as_mut().unwrap());
220
221        // Insert it into the DoryenRootConsole resource
222        let mut doryen_root_console = self.bevy_app.world_mut().resource_mut::<RootConsole>();
223        doryen_root_console.0 = self.swap_console.take();
224    }
225
226    #[inline]
227    fn restore_root_console_ownership(&mut self, api: &mut dyn DoryenApi) {
228        use std::mem::swap;
229
230        // Take the root console out of the DoryenRootConsole resource
231        let mut doryen_root_console = self
232            .bevy_app
233            .world_mut()
234            .get_resource_mut::<RootConsole>()
235            .unwrap();
236        self.swap_console = doryen_root_console.0.take();
237
238        // Hand ownership of the Doryen root console back to Doryen
239        swap(api.con(), self.swap_console.as_mut().unwrap());
240    }
241
242    #[inline]
243    fn handle_input(&mut self, api: &mut dyn DoryenApi) {
244        let mut doryen_input = self
245            .bevy_app
246            .world_mut()
247            .get_resource_mut::<Input>()
248            .unwrap();
249        let input = api.input();
250        doryen_input.handle_input(&self.mouse_button_listeners, input);
251    }
252}

254impl Engine for DoryenPluginEngine {
255    fn update(&mut self, api: &mut dyn DoryenApi) -> Option<UpdateEvent> {
256        let mut doryen_fps_info = self.bevy_app.world_mut().resource_mut::<FpsInfo>();
257        doryen_fps_info.fps = api.fps();
258        doryen_fps_info.average_fps = api.average_fps();
259
260        self.handle_input(api);
261
262        self.take_root_console_ownership(api);
263        self.bevy_app.update();
264        self.restore_root_console_ownership(api);
265
266        // Process the latest AppExit event
267        if let Some(app_exit_events) = self
268            .bevy_app
269            .world_mut()
270            .get_resource_mut::<Events<AppExit>>()
271        {
272            if self
273                .app_exit_event_reader
274                .read(&app_exit_events)
275                .last()
276                .is_some()
277            {
278                return Some(UpdateEvent::Exit);
279            }
280        }
281
282        // Process the latest SetFontPath event
283        let doryen_set_font_path_events = self
284            .bevy_app
285            .world_mut()
286            .resource_mut::<Events<SetFontPath>>();
287        if let Some(doryen_set_font_path) = self
288            .set_font_path_event_reader
289            .read(&doryen_set_font_path_events)
290            .last()
291        {
292            api.set_font_path(doryen_set_font_path.0.as_ref());
293        }
294
295        // Process the latest Capture event
296        let doryen_capture_events = self.bevy_app.world_mut().resource_mut::<Events<Capture>>();
297        if let Some(doryen_capture) = self
298            .capture_event_reader
299            .read(&doryen_capture_events)
300            .last()
301        {
302            return Some(UpdateEvent::Capture(doryen_capture.0.to_string()));
303        }
304
305        None
306    }
307
308    fn render(&mut self, api: &mut dyn DoryenApi) {
309        self.take_root_console_ownership(api);
310        self.bevy_app.world_mut().run_schedule(MainRender);
311        self.restore_root_console_ownership(api);
312    }
313
314    fn resize(&mut self, api: &mut dyn DoryenApi) {
315        let (previous_width, previous_height) = self.previous_screen_size;
316        let (new_width, new_height) = api.get_screen_size();
317
318        let mut resized_events = self
319            .bevy_app
320            .world_mut()
321            .get_resource_mut::<Events<Resized>>()
322            .unwrap();
323        let resized = Resized {
324            previous_width,
325            previous_height,
326            new_width,
327            new_height,
328        };
329        resized_events.send(resized);
330        //drop(resized_events);
331
332        match self.resize_mode {
333            ResizeMode::Nothing => (),
334            ResizeMode::Automatic => {
335                let (previous_console_width, previous_console_height) = self.previous_console_size;
336
337                let w_ratio = previous_width / previous_console_width;
338                let h_ratio = previous_height / previous_console_height;
339
340                let new_console_width = new_width / w_ratio;
341                let new_console_height = new_height / h_ratio;
342                api.con().resize(new_console_width, new_console_height);
343            }
344            ResizeMode::Callback(callback) => {
345                self.take_root_console_ownership(api);
346                callback(
347                    &mut self.bevy_app.world_mut().get_resource_mut().unwrap(),
348                    resized,
349                );
350                self.restore_root_console_ownership(api);
351            }
352        }
353
354        self.previous_screen_size = (new_width, new_height);
355        self.previous_console_size = api.con().get_size();
356    }
357}

359fn doryen_runner(mut app: BevyApp) -> AppExit {
360    let mut resource_settings = app
361        .world_mut()
362        .get_resource_or_insert_with(DoryenPluginSettings::default);
363    let DoryenPluginSettings {
364        app_options,
365        mouse_button_listeners,
366        resize_mode,
367    } = std::mem::take(&mut *resource_settings);
368
369    let AppOptions {
370        screen_height,
371        screen_width,
372        console_height,
373        console_width,
374        ..
375    } = app_options;
376
377    let mut doryen_app = DoryenApp::new(app_options);
378
379    doryen_app.set_engine(Box::new(DoryenPluginEngine {
380        bevy_app: app,
381        app_exit_event_reader: ManualEventReader::default(),
382        set_font_path_event_reader: ManualEventReader::default(),
383        capture_event_reader: ManualEventReader::default(),
384        swap_console: Some(Console::new(0, 0)),
385        mouse_button_listeners,
386        previous_screen_size: (screen_width, screen_height),
387        previous_console_size: (console_width, console_height),
388        resize_mode,
389    }));
390
391    doryen_app.run();
392
393    AppExit::Success
394}

445/// How the [`DoryenPlugin`] reacts to the resize event from Doryen.
446#[derive(Clone, Copy)]
447pub enum ResizeMode {
448    /// Do nothing when the window is resized. This is the default behavior.
449    Nothing,
450    /// Set the console size to match the window size automatically. This
451    /// retains the ratio defined between the console size and the screen size
452    /// as given in the [`AppOptions`] at the start of the program.
453    Automatic,
454    /// Call the given function when the resize event is triggered. Because
455    /// Doryen is sensitive to when the root console is resized, the safest
456    /// place to make a call to do so and always have the correct behavior is
457    /// during this resize callback which comes directly from Doryen itself. The
458    /// [`Resized`] event is useful for reacting to resizing within Bevy
459    /// systems for other reasons, but will arrive at a point that is too late
460    /// to do the root console resizing correctly.
461    Callback(fn(&mut RootConsole, Resized)),
462}
463
464impl std::fmt::Debug for ResizeMode {
465    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
466        match self {
467            Self::Nothing => f.write_str("Nothing"),
468            Self::Automatic => f.write_str("Automatic"),
469            Self::Callback(_) => f.write_str("Callback"),
470        }
471    }
