9#[derive(Component, Default)]
10struct Circle;

37pub fn main() {
38    App::new()
39        .insert_resource(DoryenPluginSettings {
40            app_options: AppOptions {

44            ..Default::default()

47        .add_systems(Startup, init)
48        .add_systems(Update, update_circle)
49        .add_systems(Render, render)
50        .run();

53fn init(mut commands: Commands) {
54    let circle = commands
55        .spawn(CircleBundle {
56            circle: Circle,
57            position: Position { x: 0., y: 0. },
58            radius: Radius(10.),
59            angle: Angle(0.),
60        })
61        .id();
62    commands.insert_resource(Entities { circle });
63}

65fn update_circle(
66    root_console: Res<RootConsole>,
67    entities: Res<Entities>,
68    mut circle_query: Query<(&mut Position, &mut Radius, &mut Angle, &Circle)>,
69) {
70    let (mut position, mut radius, mut angle, _) = circle_query.get_mut(entities.circle).unwrap();
71
72    // update the circle radius and center position
73    angle.0 += 0.6;
74    radius.0 = 10.0 + 3.0 * (angle.0 / 10.0).sin();
75    let cs = (angle.0 / 20.0).cos();
76    let sn = (angle.0 / 15.0).sin();
77    position.x = (root_console.get_width() / 2) as f32 + cs * 15.0;
78    position.y = (root_console.get_height() / 2) as f32 + sn * 15.0;
79}

81fn render(
82    entities: Res<Entities>,
83    mut root_console: ResMut<RootConsole>,
84    circle_query: Query<(&Position, &Radius, &Angle, &Circle)>,
85) {
86    // fill the console with transparent black. The more opaque it is, the faster the previous frames will fade to black.
87    // replace alpha with a lower value, like 10 or 5 and the effect will last longer.
88    root_console.clear(None, Some((0, 0, 0, 20)), None);
89    let (position, radius, angle, _) = circle_query.get(entities.circle).unwrap();
90    // here we render current frame (only a circle of blue dots)
91    for r in 0..10 {
92        let angle = angle.0 + r as f32 * std::f32::consts::PI * 2.0 / 10.0;
93        let cs = angle.cos();
94        let sn = angle.sin();
95        let x = position.x + radius.0 * cs;
96        let y = position.y + radius.0 * sn;
97        root_console.back(x as i32, y as i32, (0, 0, 255, 255));
98    }
99}
